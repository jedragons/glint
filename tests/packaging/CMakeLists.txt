cmake_minimum_required(VERSION 3.25)
project(GlintConsumer LANGUAGES C)
foreach(var CMAKE_GENERATOR_PLATFORM CMAKE_GENERATOR_TOOLSET CMAKE_VS_PLATFORM_NAME CMAKE_OSX_ARCHITECTURES)
    if(DEFINED GLINT_OUTER_${var} AND NOT "${${var}}" STREQUAL "${GLINT_OUTER_${var}}")
        message(FATAL_ERROR "Consumer ${var} '${${var}}' differs from outer build '${GLINT_OUTER_${var}}'")
    endif()
endforeach()
find_package(Glint CONFIG REQUIRED)
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE Glint::glint)
add_executable(entry_consumer entry.c)
target_link_libraries(entry_consumer PRIVATE Glint::main)
